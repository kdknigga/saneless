3. Add a `model_validator(mode="after")` to the `Settings` class that validates `tmp_dir` and `consume_dir` writability (per D-12, D-13). Implementation:

```python
import os
from pydantic import model_validator

@model_validator(mode="after")
def validate_writable_dirs(self) -> Settings:
    """Fail fast if tmp_dir or consume_dir are not writable."""
    tmp = Path(self.output.tmp_dir)
    if tmp.exists() and not os.access(tmp, os.W_OK):
        msg = f"tmp_dir is not writable: {tmp}"
        raise ValueError(msg)
    elif not tmp.exists():
        parent = tmp.parent
        if parent.exists() and not os.access(parent, os.W_OK):
            msg = f"tmp_dir parent is not writable: {parent}"
            raise ValueError(msg)
    consume = self.paperless.consume_dir
    if consume:
        consume_path = Path(consume)
        if consume_path.exists() and not os.access(consume_path, os.W_OK):
            msg = f"consume_dir is not writable: {consume_path}"
            raise ValueError(msg)
        elif not consume_path.exists():
            parent = consume_path.parent
            if parent.exists() and not os.access(parent, os.W_OK):
                msg = f"consume_dir parent is not writable: {parent}"
                raise ValueError(msg)
    return self
```

Add `import os` at the top of the file (standard library import). Add `model_validator` to the pydantic import line.
