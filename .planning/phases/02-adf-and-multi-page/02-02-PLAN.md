3. Add a method `_scan_adf_pages` to `SaneBackend`:
   ```python
   def _scan_adf_pages(self, dev: SaneDevice) -> Iterator[Image.Image]:
       """Yield validated pages from ADF via multi_scan()."""
       iterator = dev.multi_scan()
       page_num = 0
       try:
           while True:
               try:
                   page_image = next(iterator)
               except StopIteration:
                   break
               except Exception as exc:
                   error_str = str(exc).lower()
                   if page_num == 0:
                       # First page failed -- empty feeder
                       if "out of documents" in error_str or "no docs" in error_str:
                           raise FeederEmptyError("No paper detected in feeder") from exc
                       # Also catch StopIteration-like on first page
                       raise FeederEmptyError("No paper detected in feeder") from exc
                   # After first page, end-of-feed signals
                   if "out of documents" in error_str or "no docs" in error_str:
                       break
                   raise

               page_num += 1
               # Validate: nonzero dimensions
               if page_image.size[0] == 0 or page_image.size[1] == 0:
                   logger.warning("Page %d: zero dimensions, skipping", page_num)
                   continue
               # Strip EXIF (Pitfall #5: invalid EXIF breaks img2pdf)
               page_image.info.pop("exif", None)
               yield page_image
       finally:
           # Delete iterator before cancel to avoid __del__ issues (Pitfall #1 from research)
           del iterator

       if page_num == 0:
           raise FeederEmptyError("No paper detected in feeder")
   ```
